// Firestore Security Rules for 학생실습관리
// Deploy to Firebase Console: Firestore > Rules
//
// 설계 원칙: 이 앱의 모든 화면(학생 조회 키오스크 포함)은 관리자가 Google 로그인한
// 세션 안에서만 사용된다. 별도의 학생 로그인은 없으므로 "본인만 읽기" 같은 규칙은
// 불필요하고, 오히려 혼란만 준다 (예전 버전에는 그런 시도가 있었으나 실제로는
// 작동하지 않았다). 따라서 전부 관리자 전용으로 단순화한다.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
             request.auth.token.email == 'goodojb@gmail.com';
    }

    // ==================== Terms (학기) ====================

    match /terms/{termId} {
      allow read, write: if isAdmin();

      match /enrollments/{studentId} {
        allow read, write: if isAdmin();
      }

      match /scoreRecords/{recordId} {
        allow read, write: if isAdmin();
      }

      match /rouletteHistory/{id} {
        allow read, write: if isAdmin();
      }

      match /gradeImports/{id} {
        allow read, write: if isAdmin();
      }

      match /quizzes/{quizId} {
        allow read, write: if isAdmin();

        match /responses/{studentId} {
          allow read, write: if isAdmin();
        }
      }

      match /quizTemplates/{templateId} {
        allow read, write: if isAdmin();
      }
    }

    // ==================== Settings (전역) ====================

    match /settings/{document=**} {
      allow read, write: if isAdmin();
    }

    // ==================== Admins (참고용) ====================

    match /admins/{adminId} {
      allow read, write: if isAdmin();
    }

    // ==================== Deny All ====================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
